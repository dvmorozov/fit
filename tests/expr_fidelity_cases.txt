# Pascal->Python (numpy) expression-translation fidelity cases.
#
# Single source of truth for two differential tests that TOGETHER guarantee the
# translation is correct:
#   * testcase_expr_fidelity.pas (Pascal): asserts ExpressionToNumpy(pascal) == numpy
#     AND that the native engine (native_math_expr.ParseAndCalcExpression) evaluates
#     `pascal` to `expected`.
#   * test_fit_backend.py (Python): asserts the sidecar's Model(numpy) evaluates to
#     the same `expected`.
# Chain: native(pascal) == expected == numpy(numpy_of_pascal)  =>  translation preserves value.
#
# Values were generated once from the native engine (see the generator in the commit
# that introduced this file); the Pascal test re-verifies them every run, so a change
# in either the transpiler or fpexprpars is caught.
#
# Format (four fields, ' ;; ' separated):  pascal ;; numpy ;; name=val,... ;; expected
# Lines starting with '#' and blank lines are ignored.

# --- operators: power precedence and associativity, unary minus ---
x^2 ;; x**2 ;; x=3 ;; 9.000000000
2^3^2 ;; 2**3**2 ;;  ;; 512.000000000
-3^2 ;; -3**2 ;;  ;; -9.000000000
2^-1 ;; 2**-1 ;;  ;; 0.500000000
2*3^2 ;; 2*3**2 ;;  ;; 18.000000000
a^b ;; a**b ;; a=2,b=10 ;; 1024.000000000
sin(x)^2+cos(x)^2 ;; sin(x)**2+cos(x)**2 ;; x=0.9 ;; 1.000000000
x/w ;; x/w ;; x=7,w=2 ;; 3.500000000
2.5*x+3 ;; 2.5*x+3 ;; x=4 ;; 13.000000000

# --- functions whose numpy name or base differs from fpexprpars ---

log(x) ;; log10(x) ;; x=1000 ;; 3.000000000
sqr(x) ;; square(x) ;; x=5 ;; 25.000000000
ln(sqr(x)+1) ;; log(square(x)+1) ;; x=3 ;; 2.302585093

# --- functions that pass through unchanged ---
exp(0) ;; exp(0) ;;  ;; 1.000000000
sqrt(x) ;; sqrt(x) ;; x=16 ;; 4.000000000
abs(0-7) ;; abs(0-7) ;;  ;; 7.000000000
sin(x) ;; sin(x) ;; x=0.5 ;; 0.479425539
cos(x) ;; cos(x) ;; x=0.5 ;; 0.877582562
arctan(1)*4 ;; arctan(1)*4 ;;  ;; 3.141592654

# --- constant, case-insensitivity, identifier-that-starts-with-a-function-name ---
pi ;; pi ;;  ;; 3.141592654
LN(x) ;; log(x) ;; x=1 ;; 0.000000000
EXP(x) ;; exp(x) ;; x=0 ;; 1.000000000
Sqrt(x) ;; sqrt(x) ;; x=9 ;; 3.000000000
expA ;; expA ;; expA=42 ;; 42.000000000

# --- realistic curve formulas (peak and its half-max shoulder) ---
A*exp(-((x-x0)/w)^2) ;; A*exp(-((x-x0)/w)**2) ;; A=100,x=10,x0=10,w=1.5 ;; 100.000000000
A*exp(-((x-x0)/w)^2) ;; A*exp(-((x-x0)/w)**2) ;; A=100,x=11.5,x0=10,w=1.5 ;; 36.787944117

# --- Pearson VII (peak-normalised, sigma = FWHM, m = shape) ---
A/(1+(2^(1/m)-1)*(2*(x-x0)/sigma)^2)^m ;; A/(1+(2**(1/m)-1)*(2*(x-x0)/sigma)**2)**m ;; A=100,x=12,x0=10,sigma=2,m=2 ;; 14.166564366
A/(1+(2^(1/m)-1)*(2*(x-x0)/sigma)^2)^m ;; A/(1+(2**(1/m)-1)*(2*(x-x0)/sigma)**2)**m ;; A=80,x=10.7,x0=10,sigma=3,m=3 ;; 67.819091419

# --- Moffat (peak-normalised, sigma = core width, m = beta) ---
A/(1+((x-x0)/sigma)^2)^m ;; A/(1+((x-x0)/sigma)**2)**m ;; A=100,x=12,x0=10,sigma=2,m=2.5 ;; 17.677669530
A/(1+((x-x0)/sigma)^2)^m ;; A/(1+((x-x0)/sigma)**2)**m ;; A=80,x=10.6,x0=10,sigma=1.5,m=3 ;; 51.252613883

# --- Doniach-Sunjic (asymmetric core-level line; alpha = singularity index) ---
A*cos(pi*alpha/2+(1-alpha)*arctan((x-x0)/sigma))/(sigma^2+(x-x0)^2)^((1-alpha)/2) ;; A*cos(pi*alpha/2+(1-alpha)*arctan((x-x0)/sigma))/(sigma**2+(x-x0)**2)**((1-alpha)/2) ;; A=100,x=11.5,x0=10,sigma=2,alpha=0.2 ;; 32.461239725
A*cos(pi*alpha/2+(1-alpha)*arctan((x-x0)/sigma))/(sigma^2+(x-x0)^2)^((1-alpha)/2) ;; A*cos(pi*alpha/2+(1-alpha)*arctan((x-x0)/sigma))/(sigma**2+(x-x0)**2)**((1-alpha)/2) ;; A=80,x=10.7,x0=10,sigma=1.5,alpha=0.35 ;; 38.755707538

# --- special functions erf/erfc/erfcx (mirrored from scipy.special) ---
erf(x) ;; erf(x) ;; x=0.75 ;; 0.711155634

# --- Exponentially modified Gaussian (stable emg primitive; A = area) ---
A*emg(x-x0,sigma,tau) ;; A*emg(x-x0,sigma,tau) ;; A=100,x=10,x0=10,sigma=1,tau=0.8 ;; 28.845040163
A*emg(x-x0,sigma,tau) ;; A*emg(x-x0,sigma,tau) ;; A=80,x=12,x0=10,sigma=1.5,tau=2 ;; 14.039226049

# --- true Voigt (Faddeeva; A = area, sigma Gaussian std, gamma Lorentzian HWHM) ---
A*voigt(x-x0,sigma,gamma) ;; A*voigt(x-x0,sigma,gamma) ;; A=100,x=11.5,x0=10,sigma=1,gamma=0.5 ;; 13.543427630
A*voigt(x-x0,sigma,gamma) ;; A*voigt(x-x0,sigma,gamma) ;; A=80,x=12.5,x0=10,sigma=2,gamma=1 ;; 6.692777644

# --- Skewed Gaussian (Gaussian x erf skew; A = area scale, beta = skew) ---
A/(sigma*sqrt(2*pi))*exp(-(x-x0)^2/(2*sigma^2))*(1+erf(beta*(x-x0)/(sigma*sqrt(2)))) ;; A/(sigma*sqrt(2*pi))*exp(-(x-x0)**2/(2*sigma**2))*(1+erf(beta*(x-x0)/(sigma*sqrt(2)))) ;; A=100,x=10.7,x0=10,sigma=1,beta=2 ;; 57.407469775
A/(sigma*sqrt(2*pi))*exp(-(x-x0)^2/(2*sigma^2))*(1+erf(beta*(x-x0)/(sigma*sqrt(2)))) ;; A/(sigma*sqrt(2*pi))*exp(-(x-x0)**2/(2*sigma**2))*(1+erf(beta*(x-x0)/(sigma*sqrt(2)))) ;; A=80,x=9.5,x0=10,sigma=1.5,beta=-1 ;; 25.382641311

# --- Step (erf edge; A = height, x0 = edge centre, sigma = width) ---
(A/2)*(1+erf((x-x0)/(sigma*sqrt(2)))) ;; (A/2)*(1+erf((x-x0)/(sigma*sqrt(2)))) ;; A=100,x=11,x0=10,sigma=0.5 ;; 97.724986805
(A/2)*(1+erf((x-x0)/(sigma*sqrt(2)))) ;; (A/2)*(1+erf((x-x0)/(sigma*sqrt(2)))) ;; A=80,x=10.3,x0=10,sigma=1 ;; 49.432913775

# Ten are renamed to their numpy spelling; the six numpy has no name for keep

# is pi/2 - arctan rather than arctan(1/x) (which is discontinuous at zero), and
